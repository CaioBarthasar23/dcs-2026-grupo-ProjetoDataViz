### Comando de construção e consulta

Comando executado a partir de `projeto_final/`; cria a Silver fixture e a Gold em `TemporaryDirectory` local, carrega as seis tabelas Parquet no DuckDB e imprime as linhas por tabela e os resultados das duas consultas:

```bash
python - <<'PY'
from pathlib import Path
from tempfile import TemporaryDirectory

import duckdb
import io
import pyarrow.parquet as pq

from src.silver import LocalLake
from tests.test_gold_schema import GOLD_TABLES, gold_module, silver_fixture

sql_orfas = """SELECT 'produto' AS relacao, count(*) AS orfas
FROM fato_preco_diario f LEFT JOIN dim_produto p USING (group_id, product_id)
WHERE p.product_id IS NULL
UNION ALL
SELECT 'colecao', count(*)
FROM fato_preco_diario f LEFT JOIN dim_colecao c USING (group_id)
WHERE c.group_id IS NULL
UNION ALL
SELECT 'data_preco', count(*)
FROM fato_preco_diario f LEFT JOIN dim_data d USING (data)
WHERE d.data IS NULL
UNION ALL
SELECT 'acabamento', count(*)
FROM fato_preco_diario f LEFT JOIN dim_acabamento a USING (acabamento)
WHERE a.acabamento IS NULL
UNION ALL
SELECT 'data_cambio', count(*)
FROM fato_cambio_diario f LEFT JOIN dim_data d USING (data)
WHERE d.data IS NULL
ORDER BY relacao"""
sql_asof = """SELECT data, product_id, data_ptax_usada, market_price_brl
FROM fato_preco_diario
ORDER BY data, product_id"""

with TemporaryDirectory(prefix="t11-gold-") as tmp:
    root = Path(tmp)
    fonte = silver_fixture(root / "silver")
    destino = LocalLake(root / "gold")
    counts = gold_module().build(fonte, destino)

    con = duckdb.connect()
    for table in GOLD_TABLES:
        key = f"gold/group_id=23651/{table}.parquet"
        con.register(table, pq.read_table(io.BytesIO(destino.get(key))))

    print("linhas_por_tabela:")
    for table, count in sorted(counts.items()):
        print(f"{table}: rows={count}")
    print("consulta_chaves_orfas:")
    print(sql_orfas)
    print("chaves_orfas_por_relacao=", con.execute(sql_orfas).fetchall())
    print("consulta_asof:")
    print(sql_asof)
    print("preco_asof=", con.execute(sql_asof).fetchall())
    con.close()
PY
```
